function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = e.parameter;
  var timestamp = new Date();

  sheet.appendRow([
    timestamp,
    data.name || '',
    data.phone || '',
    data.email || '',
    data.event_date || '',
    data.venue || '',
    data.package || '',
    data.guest_count || '',
    data.message || ''
  ]);

  var notifyEmail = 'barrettjessen21@gmail.com';
  var subject = 'New website lead: ' + (data.name || 'Unknown') + ' — Bloomkeep Co.';
  var body =
    'New inquiry from the Bloomkeep Co. website:\n\n' +
    'Name: ' + (data.name || '') + '\n' +
    'Phone: ' + (data.phone || '') + '\n' +
    'Email: ' + (data.email || '') + '\n' +
    'Event Date: ' + (data.event_date || '') + '\n' +
    'Venue/City: ' + (data.venue || '') + '\n' +
    'Package Interest: ' + (data.package || '') + '\n' +
    'Guest Count: ' + (data.guest_count || '') + '\n' +
    'Message: ' + (data.message || '') + '\n\n' +
    'Submitted: ' + timestamp;

  MailApp.sendEmail(notifyEmail, subject, body);

  return ContentService
    .createTextOutput(JSON.stringify({ result: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
